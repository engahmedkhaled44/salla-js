// زر الانتقال للأعلى
var scrollTopFloat = document.createElement('div');
scrollTopFloat.className = 'scroll-top-float';

var scrollTopButton = document.createElement('button');
scrollTopButton.className = 'scroll-top-button';

var scrollTopIcon = document.createElement('img');
scrollTopIcon.src = 'https://i.top4top.io/p_3336jr2tc1.png';
scrollTopIcon.alt = 'Scroll to Top';
scrollTopIcon.className = 'scroll-top-icon';

scrollTopButton.appendChild(scrollTopIcon);
scrollTopFloat.appendChild(scrollTopButton);
document.body.appendChild(scrollTopFloat);

scrollTopButton.addEventListener('click', function () {
    window.scrollTo({ top: 0, behavior: 'smooth' });
});

// زر الانتقال للأسفل
var scrollBottomFloat = document.createElement('div');
scrollBottomFloat.className = 'scroll-bottom-float';

var scrollBottomButton = document.createElement('button');
scrollBottomButton.className = 'scroll-bottom-button';

var scrollBottomIcon = document.createElement('img');
scrollBottomIcon.src = 'https://d.top4top.io/p_3336sf46s1.png';
scrollBottomIcon.alt = 'Scroll to Bottom';
scrollBottomIcon.className = 'scroll-bottom-icon';

scrollBottomButton.appendChild(scrollBottomIcon);
scrollBottomFloat.appendChild(scrollBottomButton);
document.body.appendChild(scrollBottomFloat);

scrollBottomButton.addEventListener('click', function () {
    window.scrollTo({ top: document.body.scrollHeight, behavior: 'smooth' });
});

// إضافة الـ CSS باستخدام JavaScript
var style = document.createElement('style');
style.innerHTML = `
    .scroll-top-float {
        position: fixed;
        bottom: 50px;
        right: 30px;
        z-index: 100;
    }

    .scroll-bottom-float {
        position: fixed;
        bottom: 50px;
        right: 100px;
        z-index: 100;
    }

    @keyframes shadowPulse {
        0% { box-shadow: 0 0 20px #A24258; }
        25% { box-shadow: 0 0 20px #A24258; }
        50% { box-shadow: 0 0 20px #442D52; }
        75% { box-shadow: 0 0 20px #442D52; }
        100% { box-shadow: 0 0 10px #A24258; }
    }

    .scroll-top-button, .scroll-bottom-button {
        background: transparent;
        border: none;
        cursor: pointer;
        padding: 0;
    }

    .scroll-top-icon, .scroll-bottom-icon {
        animation: shadowPulse 3s infinite alternate;
        width: 60px;
        height: 60px;
        border-radius: 50%;
        box-shadow: 0 4px 8px #442D52;
        cursor: pointer;
        transition: transform 0.3s ease-in-out;
    }

    .scroll-top-icon:hover, .scroll-bottom-icon:hover {
        transform: scale(1.1);
    }
`;
document.head.appendChild(style);
